RAII e Smart Pointers (unique_ptr / shared_ptr)

Gerenciamento automático de recursos sem vazamentos.

#include <memory>
#include <iostream>

struct Conexao {
    ~Conexao() { std::cout << "Fechada automaticamente\n"; }
};

int main() {
    auto ptr = std::make_unique<Conexao>();
    auto compartilhado = std::make_shared<int>(42);
}
Templates, Concepts e std::ranges

Restrições de tipo em compilação e pipeline funcional.

#include <concepts>
#include <vector>
#include <ranges>

template <typename T>
concept Numerico = std::integral<T> || std::floating_point<T>;

template <Numerico T>
T somar(T a, T b) { return a + b; }

std::vector<int> v = {1, 2, 3, 4, 5, 6};
auto pares = v | std::views::filter([](int n) { return n % 2 == 0; });

2. STL Moderna: unordered_map, optional e expected

Tabelas Hash (std::unordered_map) & Structured Bindings

Acesso $O(1)$ e desestruturação limpa de chave e valor.

#include <unordered_map>
#include <string>
#include <print>

std::unordered_map<std::string, int> notas = {{"Ana", 10}, {"Leo", 9}};
for (const auto& [aluno, nota] : notas) {
    std::println("{}: {}", aluno, nota);
}
Tratamento Funcional com std::expected

Retorno explícito de valor ou erro sem exceções.

#include <expected>
#include <string>

std::expected<double, std::string> dividir(double a, double b) {
    if (b == 0.0) return std::unexpected("Divisão por zero");
    return a / b;
}
